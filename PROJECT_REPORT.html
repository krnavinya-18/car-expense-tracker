AIM:

Development of Car Expense Tracking System, a comprehensive Java-based browser application for monitoring, categorizing, and analyzing automobile running expenditures using Java 21, Spring Boot, Vaadin Flow, Oracle Database, Spring Data JPA, and Apache Maven.

OBJECTIVE:

To design and implement a modern, responsive, and robust enterprise web system that enables vehicle owners and fleet managers to systematically record vehicle operational costs across various categories (Fuel, Maintenance, Repairs, Insurance, Parking, Tolls, Car Wash, and Taxes). The system provides a live financial dashboard with instant key performance metrics, a full-featured management grid with multi-parameter search, category and date filtering, inline modal editing, secure deletion, and visual cost breakdown charts with percentage distributions. All frontend components and backend services are written purely in Java using Vaadin Flow, backed by Oracle Database persistence and safe ORM transactions.

SYSTEM ARCHITECTURE:

The Car Expense Tracking System follows a clean, layered enterprise architecture: Model – Repository – Service – View. A client browser request triggers Vaadin Flow’s server-driven engine. The active View invokes the Service layer to execute business rules and query validations. The Service layer communicates with the Spring Data JPA Repository, which translates operations into safe, parameterized SQL executed against the Oracle Database. Results are bound directly to Java component models, and Vaadin automatically synchronizes the client DOM without manual JavaScript or HTML generation.

MODEL LAYER – Expense.java:

The Expense class maps directly to the CAR_EXPENSES table in Oracle Database.

package com.carexpense.model;

import jakarta.persistence.*;
import java.math.BigDecimal;
import java.time.LocalDate;

@Entity
@Table(name = "CAR_EXPENSES")
public class Expense {

    @Id
    @GeneratedValue(strategy = GenerationType.SEQUENCE, generator = "car_expenses_seq")
    @SequenceGenerator(name = "car_expenses_seq", sequenceName = "CAR_EXPENSES_SEQ", allocationSize = 1)
    @Column(name = "EXPENSE_ID")
    private Long id;

    @Column(name = "EXPENSE_DATE", nullable = false)
    private LocalDate expenseDate;

    @Column(name = "CATEGORY", nullable = false, length = 100)
    private String category;

    @Column(name = "AMOUNT", nullable = false, precision = 12, scale = 2)
    private BigDecimal amount;

    @Column(name = "DESCRIPTION", length = 500)
    private String description;

    public Expense() {}

    public Expense(LocalDate expenseDate, String category, BigDecimal amount, String description) {
        this.expenseDate = expenseDate;
        this.category = category;
        this.amount = amount;
        this.description = description;
    }

    public Long getId() { return id; }
    public void setId(Long id) { this.id = id; }
    public LocalDate getExpenseDate() { return expenseDate; }
    public void setExpenseDate(LocalDate expenseDate) { this.expenseDate = expenseDate; }
    public String getCategory() { return category; }
    public void setCategory(String category) { this.category = category; }
    public BigDecimal getAmount() { return amount; }
    public void setAmount(BigDecimal amount) { this.amount = amount; }
    public String getDescription() { return description; }
    public void setDescription(String description) { this.description = description; }
}
REPOSITORY LAYER – ExpenseRepository.java:

Provides CRUD methods and parameterized JPQL aggregate queries for reporting and dashboard statistics.

package com.carexpense.repository;

import com.carexpense.model.Expense;
import org.springframework.data.domain.Pageable;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import org.springframework.stereotype.Repository;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.List;

@Repository
public interface ExpenseRepository extends JpaRepository<Expense, Long> {

    List<Expense> findByCategoryIgnoreCase(String category);

    @Query("SELECT e FROM Expense e WHERE " +
           "LOWER(e.category) LIKE LOWER(CONCAT('%', :term, '%')) OR " +
           "LOWER(e.description) LIKE LOWER(CONCAT('%', :term, '%'))")
    List<Expense> searchByCategoryOrDescription(@Param("term") String term);

    List<Expense> findByExpenseDateBetween(LocalDate startDate, LocalDate endDate);

    @Query("SELECT e FROM Expense e WHERE (:category IS NULL OR e.category = :category) AND " +
           "(:term IS NULL OR LOWER(e.category) LIKE LOWER(CONCAT('%', :term, '%')) " +
           "OR LOWER(e.description) LIKE LOWER(CONCAT('%', :term, '%')))")
    List<Expense> findByCategoryAndSearchTerm(@Param("category") String category, @Param("term") String term);

    @Query("SELECT COALESCE(SUM(e.amount), 0) FROM Expense e")
    BigDecimal getTotalExpenses();

    @Query("SELECT COALESCE(AVG(e.amount), 0) FROM Expense e")
    BigDecimal getAverageExpense();

    @Query("SELECT e.category FROM Expense e GROUP BY e.category ORDER BY SUM(e.amount) DESC")
    List<String> findHighestSpendingCategory();

    @Query("SELECT e FROM Expense e ORDER BY e.expenseDate DESC, e.id DESC")
    List<Expense> findTop5RecentExpenses(Pageable pageable);

    @Query("SELECT e.category, SUM(e.amount) FROM Expense e GROUP BY e.category ORDER BY SUM(e.amount) DESC")
    List<Object[]> getCategoryWiseSpending();

    @Query("SELECT EXTRACT(YEAR FROM e.expenseDate), EXTRACT(MONTH FROM e.expenseDate), SUM(e.amount) " +
           "FROM Expense e GROUP BY EXTRACT(YEAR FROM e.expenseDate), EXTRACT(MONTH FROM e.expenseDate) " +
           "ORDER BY EXTRACT(YEAR FROM e.expenseDate) DESC, EXTRACT(MONTH FROM e.expenseDate) DESC")
    List<Object[]> getMonthlySpendingSummary();
}
SERVICE LAYER – ExpenseService.java (Core Methods):

Coordinates business transactions, validates input parameters, and filters expense data.

package com.carexpense.service;

import com.carexpense.model.Expense;
import com.carexpense.repository.ExpenseRepository;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Sort;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.math.BigDecimal;
import java.util.List;

@Service
@Transactional
public class ExpenseService {

    private final ExpenseRepository expenseRepository;

    public ExpenseService(ExpenseRepository expenseRepository) {
        this.expenseRepository = expenseRepository;
    }

    public Expense saveExpense(Expense expense) {
        validateExpense(expense);
        return expenseRepository.save(expense);
    }

    @Transactional(readOnly = true)
    public List<Expense> getAllExpenses() {
        return expenseRepository.findAll(Sort.by(Sort.Direction.DESC, "expenseDate", "id"));
    }

    public void deleteExpense(Long id) {
        expenseRepository.deleteById(id);
    }

    @Transactional(readOnly = true)
    public List<Expense> searchExpenses(String category, String searchTerm) {
        boolean hasCat = category != null && !category.isBlank() && !category.equals("All Categories");
        boolean hasTerm = searchTerm != null && !searchTerm.isBlank();

        if (hasCat && hasTerm) return expenseRepository.findByCategoryAndSearchTerm(category, searchTerm);
        if (hasCat) return expenseRepository.findByCategoryIgnoreCase(category);
        if (hasTerm) return expenseRepository.searchByCategoryOrDescription(searchTerm);
        return getAllExpenses();
    }

    @Transactional(readOnly = true)
    public BigDecimal getTotalExpenses() { return expenseRepository.getTotalExpenses(); }

    @Transactional(readOnly = true)
    public BigDecimal getAverageExpense() { return expenseRepository.getAverageExpense(); }

    @Transactional(readOnly = true)
    public long getTotalCount() { return expenseRepository.count(); }

    @Transactional(readOnly = true)
    public String getHighestSpendingCategory() {
        List<String> cats = expenseRepository.findHighestSpendingCategory();
        return (cats != null && !cats.isEmpty()) ? cats.get(0) : "N/A";
    }

    @Transactional(readOnly = true)
    public List<Expense> getRecentExpenses() {
        return expenseRepository.findTop5RecentExpenses(PageRequest.of(0, 5));
    }

    private void validateExpense(Expense expense) {
        if (expense.getExpenseDate() == null) throw new IllegalArgumentException("Expense date is required.");
        if (expense.getCategory() == null || expense.getCategory().isBlank()) 
            throw new IllegalArgumentException("Expense category is required.");
        if (expense.getAmount() == null || expense.getAmount().compareTo(BigDecimal.ZERO) <= 0)
            throw new IllegalArgumentException("Expense amount must be greater than zero.");
    }
}
SERVICE LAYER – ReportService.java:

Aggregates spending totals, calculates category percentages, and constructs monthly timeline maps.

package com.carexpense.service;

import com.carexpense.repository.ExpenseRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.time.Month;
import java.time.format.TextStyle;
import java.util.*;

@Service
@Transactional(readOnly = true)
public class ReportService {

    private final ExpenseRepository expenseRepository;

    public ReportService(ExpenseRepository expenseRepository) {
        this.expenseRepository = expenseRepository;
    }

    public BigDecimal getTotalSpending() {
        BigDecimal total = expenseRepository.getTotalExpenses();
        return total != null ? total : BigDecimal.ZERO;
    }

    public long getTotalExpenseCount() { return expenseRepository.count(); }

    public String getHighestSpendingCategory() {
        List<String> result = expenseRepository.findHighestSpendingCategory();
        return (result != null && !result.isEmpty()) ? result.get(0) : "N/A";
    }

    public Map<String, BigDecimal> getCategorySpendingMap() {
        List<Object[]> rawData = expenseRepository.getCategoryWiseSpending();
        Map<String, BigDecimal> result = new LinkedHashMap<>();
        for (Object[] row : rawData) {
            String category = (String) row[0];
            BigDecimal total = (row[1] instanceof BigDecimal) ? (BigDecimal) row[1]
                             : BigDecimal.valueOf(((Number) row[1]).doubleValue());
            result.put(category, total.setScale(2, RoundingMode.HALF_UP));
        }
        return result;
    }

    public Map<String, BigDecimal> getCategoryPercentageMap() {
        Map<String, BigDecimal> spending = getCategorySpendingMap();
        BigDecimal total = getTotalSpending();
        Map<String, BigDecimal> percentages = new LinkedHashMap<>();

        if (total.compareTo(BigDecimal.ZERO) == 0) {
            spending.keySet().forEach(k -> percentages.put(k, BigDecimal.ZERO));
            return percentages;
        }

        for (Map.Entry<String, BigDecimal> entry : spending.entrySet()) {
            BigDecimal pct = entry.getValue().multiply(new BigDecimal("100"))
                    .divide(total, 1, RoundingMode.HALF_UP);
            percentages.put(entry.getKey(), pct);
        }
        return percentages;
    }
}

private void handleSave() {
    if (dateField.isEmpty() || categoryField.isEmpty() || amountField.isEmpty()) {
        Notification.show("Please correct the highlighted fields before saving.", 4000, Position.TOP_END)
                    .addThemeVariants(NotificationVariant.LUMO_ERROR);
        return;
    }
    if (amountField.getValue().compareTo(BigDecimal.ZERO) <= 0) {
        Notification.show("Amount must be greater than ₹0.", 4000, Position.TOP_END)
                    .addThemeVariants(NotificationVariant.LUMO_ERROR);
        return;
    }
    Expense expense = new Expense(dateField.getValue(), categoryField.getValue(),
                                  amountField.getValue(), descriptionField.getValue().trim());
    expenseService.saveExpense(expense);
    Notification.show("Expense saved successfully! (ID: " + expense.getId() + ")", 4000, Position.TOP_END)
                .addThemeVariants(NotificationVariant.LUMO_SUCCESS);
    clearForm();
}
